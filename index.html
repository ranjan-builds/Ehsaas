Connect your Google Sheet in 2 easy steps!
  1. Create a new Google Sheet with header row: id, content, author, category, language, tags, likes, shares, featured, created_at, status
  2. Go to Extensions > Apps Script and paste the backend code below: 
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var result = [];
  for (var i = 1; i < data.length; i++) {
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = data[i][j];
    }
    if (obj.status !== 'archived') result.push(obj);
  }
  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var item = JSON.parse(e.postData.contents);
  var values = sheet.getDataRange().getValues();
  var row = -1;
  for (var i = 1; i < values.length; i++) {
    if (String(values[i][0]) === String(item.id)) {
      row = i + 1;
      break;
    }
  }

  if (item.action === 'add') {
    sheet.appendRow([item.id, item.content, item.author, item.category, item.language, item.tags, item.likes || 0, item.shares || 0, item.featured, item.created_at || new Date().toISOString(), 'active']);
  } else if (item.action === 'update' && row > 0) {
    sheet.getRange(row, 1, 1, 11).setValues([[
      item.id, item.content, item.author, item.category, item.language,
      item.tags, item.likes || 0, item.shares || 0, item.featured,
      item.created_at, 'active'
    ]]);
  } else if (item.action === 'delete' && row > 0) {
    sheet.getRange(row, 11).setValue('archived');
  } else if (item.action === 'like' && row > 0) {
    // Column G is likes. The server owns the count to prevent stale clients
    // from overwriting other visitors' likes.
    var likesCell = sheet.getRange(row, 7);
    var currentLikes = Number(likesCell.getValue()) || 0;
    var nextLikes = Math.max(0, currentLikes + (Number(item.delta) || 0));
    likesCell.setValue(nextLikes);
    return ContentService.createTextOutput(JSON.stringify({status: 'success', likes: nextLikes})).setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify({status: 'success'})).setMimeType(ContentService.MimeType.JSON);
}
